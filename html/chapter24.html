The DOM is how JavaScript talks to your HTML. Master this and you can change anything on the page. Anything. 😎

What is the DOM?

When a browser loads an HTML page it creates a model of that page in memory. This model is called the Document Object Model — or DOM. It represents every element on the page as an object, arranged in a tree structure with the document at the top.

JavaScript can read this model and change it. When JavaScript changes the DOM, the browser updates what's displayed on screen instantly — without reloading the page. This is how every interactive webpage works.

You've actually been using the DOM since Chapter 20 — every time you wrote document.getElementById() you were accessing the DOM. Now we're going to understand properly what's happening and learn more of what it can do.

The DOM Tree

The DOM represents your HTML as a tree of nodes. Every element, attribute, and piece of text is a node. The document is the root node, html sits below it, then head and body, then everything inside those.

Understanding this tree structure helps you navigate the DOM and target exactly the elements you want to change.

document // the root
  html // the html element
    head // the head element
    body // the body element
      h1 // a heading
      p // a paragraph
      div // a div
        p // a paragraph inside the div

Accessing Elements

There are several ways to select elements from the DOM. The most common ones are:

  • document.getElementById('id') — selects one element by its id. Returns a single element.
  • document.getElementsByClassName('class') — selects all elements with that class. Returns a collection.
  • document.getElementsByTagName('tag') — selects all elements of that tag type. Returns a collection.
  • document.querySelector('selector') — selects the first element that matches a CSS selector.
  • document.querySelectorAll('selector') — selects all elements that match a CSS selector. Returns a collection.
var title = document.getElementById('title');
var buttons = document.getElementsByTagName('button');
var firstGreen = document.querySelector('.green');

Reading and Changing Content

Once you have an element you can read or change its content using these properties:

  • element.textContent — gets or sets the text content of an element. Safe — treats everything as plain text.
  • element.innerHTML — gets or sets the HTML content of an element. Powerful — can insert actual HTML tags. Use carefully.
var title = document.getElementById('title');

// Read the content
console.log(title.textContent);

// Change the content
title.textContent = 'New Title';

// Insert HTML
title.innerHTML = '<em>New Title</em>';

Changing Styles

You can change the CSS styles of any element using the style property. CSS property names in JavaScript use camelCase instead of hyphens — so background-color becomes backgroundColor and font-size becomes fontSize.

var title = document.getElementById('title');

title.style.color = '#00ff88';
title.style.fontSize = '2rem';
title.style.display = 'none'; // hides the element
title.style.display = 'block'; // shows it again

Working with Classes

Rather than changing individual styles, it's often better to add or remove CSS classes. This keeps your styling in your CSS file where it belongs and lets JavaScript just toggle behaviour.

var element = document.getElementById('game');

element.classList.add('active'); // adds a class
element.classList.remove('active'); // removes a class
element.classList.toggle('active'); // adds if absent, removes if present
element.classList.contains('active'); // returns true or false

You've actually already seen classList.toggle in action — it's how the XGENFUTBOT speech bubble on this very guide works when you click the genie icon!

Creating and Adding Elements

JavaScript can create brand new HTML elements and add them to the page dynamically — without any HTML being written beforehand.

// Create a new element
var newParagraph = document.createElement('p');

// Give it some content
newParagraph.textContent = 'This paragraph was created by JavaScript!';

// Add it to the page
document.getElementById('game').appendChild(newParagraph);

appendChild adds the new element as the last child of the target element. You've already been using createElement and appendChild in the game — that's how the choice buttons get built dynamically each time showChoices() runs.

Reading Attributes

You can get and set any HTML attribute on an element using getAttribute and setAttribute.

var btn = document.getElementById('startBtn');

// Read an attribute
btn.getAttribute('id'); // 'startBtn'

// Set an attribute
btn.setAttribute('disabled', true); // disables the button

// You've already used this in the game!
choices.setAttribute('data-mode', 'question');

// ==============================
// DOM FUNCTIONS
// ==============================

function showMessage(text) {
  document.getElementById('message').textContent = text;
}

function showStory(text) {
  document.getElementById('story').textContent = text;
}

function updateHealthBar() {
  for (var i = 1; i <= 3; i++) {
    var heart = document.getElementById('heart' + i);
    if (i <= playerHealth) {
      heart.classList.remove('heart-lost');
    } else {
      heart.classList.add('heart-lost');
    }
  }
}

function updateInventory() {
  var itemsDiv = document.getElementById('inventory-items');
  itemsDiv.innerHTML = '';
  for (var i = 0; i < inventory.length; i++) {
    var item = document.createElement('span');
    item.className = 'inventory-item';
    item.textContent = inventory[i].icon + ' ' + inventory[i].name;
    itemsDiv.appendChild(item);
  }
}

function updateScore() {
  document.getElementById('score-display').textContent = 'Score: ' + playerScore;
}

function showChoices(choices) {
  var choicesDiv = document.getElementById('choices');
  choicesDiv.innerHTML = '';
  for (var i = 0; i < choices.length; i++) {
    var btn = document.createElement('button');
    btn.textContent = choices[i];
    btn.setAttribute('data-index', i);
    choicesDiv.appendChild(btn);
  }
}

// GAME FUNCTIONS
// ==============================

function startGame() {
  var nameInput = document.getElementById('nameInput').value;
  if (nameInput === '') {
    showMessage('You need to enter your name, coder!');
  } else {
    playerName = nameInput;
    document.getElementById('input-area').style.display = 'none';
    document.getElementById('health-bar').style.display = 'block';
    document.getElementById('inventory-panel').style.display = 'block';
    document.getElementById('score-display').style.display = 'block';
    updateHealthBar();
    updateScore();
    showStory('Welcome, ' + playerName + '. I am XGENFUTBOT — your guide, your conscience, and occasionally your biggest critic. The Code Machine has been dismantled and its five components scattered across five dangerous realms. You must retrieve them all. Do you have what it takes?');
    showMessage('');
    showChoices(['I was born ready. Let\'s go.', 'I\'m terrified but let\'s do this anyway.']);
    document.getElementById('choices').setAttribute('data-mode', 'intro');
  }
}

function enterRealm() {
  var realm = realms[currentRealm];
  showStory('You enter ' + realm.name + '. The ' + realm.guardian + ' blocks your path. "Answer my question and the ' + realm.reward + ' is yours," it growls.');
  showMessage(realm.question);
  showChoices(realm.options);
  document.getElementById('choices').setAttribute('data-mode', 'question');
}

function handleAnswer(index) {
  var realm = realms[currentRealm];
  if (index === realm.correct) {
    inventory.push({ name: realm.reward, icon: realm.rewardIcon });
    playerScore += 100;
    updateInventory();
    updateScore();
    currentRealm++;
    if (currentRealm < realms.length) {
      showStory('Correct! The ' + realm.reward + ' ' + realm.rewardIcon + ' is yours. XGENFUTBOT nods approvingly. "Not bad. But there are ' + (realms.length - currentRealm) + ' realm' + (realms.length - currentRealm === 1 ? '' : 's') + ' left. Don\'t celebrate yet."');
      showMessage('');
      showChoices(['Continue to the next realm']);
      document.getElementById('choices').setAttribute('data-mode', 'continue');
    } else {
      endGame();
    }
  } else {
    playerHealth--;
    updateHealthBar();
    if (playerHealth === 0) {
      gameOver();
    } else {
      showStory('Hmm — not quite. But that\'s completely fine. Nobody gets everything first time. The important thing is that now you\'ve seen the right answer, it\'ll stick. You have ' + playerHealth + ' ' + (playerHealth === 1 ? 'life' : 'lives') + ' remaining. Try again!');
      showMessage(realm.question);
      showChoices(realm.options);
      document.getElementById('choices').setAttribute('data-mode', 'question');
    }
  }
}

function gameOver() {
  showStory('Hey — don\'t feel bad about this. Seriously. The fact that you got this far means you\'re already doing brilliantly. The section that tripped you up is worth one more read — not because you\'re behind, but because once it clicks it clicks forever.');
  showMessage('XGENFUTBOT tips his fedora. "Every developer has been exactly where you are right now. Every single one."');
  showChoices(['Try again from the beginning']);
  document.getElementById('choices').setAttribute('data-mode', 'restart');
}

function endGame() {
  var ending;
  if (playerHealth === 3) {
    ending = 'PERFECT RUN! The Code Machine roars to life, all five components locked in. XGENFUTBOT does an actual victory dance. "I knew you had it. Full marks, full lives, full brilliance. You are ready for what comes next."';
  } else if (playerHealth === 2) {
    ending = 'The Code Machine activates — a few sparks fly but it works. XGENFUTBOT nods. "Solid run. One stumble does not a failure make. You\'re ready."';
  } else {
    ending = 'The Code Machine splutters to life. XGENFUTBOT raises an eyebrow. "It works. Barely. But it works. Go back and review the bits that caught you out — then you\'ll be genuinely ready."';
  }
  showStory(ending);
  showMessage('🎉 Quest Complete! Final Score: ' + playerScore);
  document.getElementById('choices').innerHTML = '';
}

// ==============================
// EVENT LISTENERS
// ==============================

document.getElementById('startBtn').addEventListener('click', startGame);

document.getElementById('choices').addEventListener('click', function(e) {
  if (e.target.tagName !== 'BUTTON') return;
  var mode = document.getElementById('choices').getAttribute('data-mode');
  var index = parseInt(e.target.getAttribute('data-index'));
  if (mode === 'intro') {
    enterRealm();
  } else if (mode === 'question') {
    handleAnswer(index);
  } else if (mode === 'continue') {
    enterRealm();
  } else if (mode === 'restart') {
    playerHealth = 3;
    playerScore = 0;
    inventory = [];
    currentRealm = 0;
    updateHealthBar();
    updateInventory();
    updateScore();
    enterRealm();
  }
});

⚡ What the DOM is doing here

Look at the updateHealthBar() function — it uses a for loop to select each heart element by id and adds or removes a CSS class depending on the player's health. That's the DOM and classList working together.

The updateInventory() function clears the inventory panel using innerHTML = '' and then uses createElement and appendChild to build new item tags dynamically every time a component is collected. The page is rebuilding part of itself in real time.

Every visual update in this game — every story change, every heart fading, every inventory item appearing — is JavaScript talking to the DOM and the DOM updating what you see on screen.
